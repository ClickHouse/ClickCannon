📍 工作目录
─────────────────────────────────────────────────────────────────────
cd /home/sunchanglong/go_project/src/ClickCannon


🚀 启动服务
─────────────────────────────────────────────────────────────────────
# 1. 启动监控系统（Grafana）
./start-monitoring.sh

# 2. 启动压测（新终端）
./clickcannon --config config.yaml

# 或者用 Go 运行
go run . --config config.yaml

  --user=admin --password='V%t^ckmstB' \

🛑 停止服务
─────────────────────────────────────────────────────────────────────
# 停止压测
按 Ctrl+C

# 停止监控
docker-compose down

# 停止并删除数据
docker-compose down -v


🔍 日志查看
─────────────────────────────────────────────────────────────────────
# 查看 Grafana 日志
docker-compose logs grafana

# 实时查看日志
docker-compose logs -f grafana

# 查看 ClickCannon 日志（如果后台运行）
tail -f clickcannon.log

─────────────────────────────────────────────────────────────────────

docker-compose restart grafana

# 查看容器状态
docker-compose ps

clickhouse-client --host=10.160.152.186 --port=9000 \

🎯 性能调优
─────────────────────────────────────────────────────────────────────
# 提高吞吐量（编辑 config.yaml）
disk:
  threads: 16
  mb_per_second_limit: 20480

insert:
  threads: 16
  batch_size: 200000

# 降低资源使用
disk:
  threads: 4
  mb_per_second_limit: 2048

insert:
  threads: 4
  batch_size: 50000


🆘 故障排查
─────────────────────────────────────────────────────────────────────
# Grafana 无法启动
sudo netstat -tlnp | grep 3000    # 检查端口占用
docker-compose logs grafana        # 查看日志

# Dashboard 无数据
./check-metrics.sh                 # 验证数据写入
# 检查 config.yaml 中 metrics.enabled: true
# 确认 Grafana 中选择了正确的 run_id

# 压测速度慢
# 1. 检查 mb_per_second_limit 是否过低
# 2. 增加 threads 数量
# 3. 增大 batch_size

# ClickHouse 连接失败
telnet 10.160.152.186 9000
clickhouse-client --host=10.160.152.186 --query="SELECT 1"
